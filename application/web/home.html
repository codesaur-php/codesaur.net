<?php            /* Raptor Framework */

namespace Web;

use Psr\Http\Message\ResponseInterface;
use Dashboard\SessionMiddleware;
use Dashboard\Localization\LocalizationMiddleware;

class Application extends \codesaur\Http\Application\Application
{
    public function __construct(ResponseInterface $response)
    {
        parent::__construct($response);

        $this->use(new Template\ExceptionHandler());
        $this->use(new SessionMiddleware(
            fn($path, $method) => str_starts_with($path, '/session/')
        ));
        $this->use(new LocalizationMiddleware('WEB_LANGUAGE_CODE'));
        $this->use(new WebRouter());
        $this->use(new Portal\PortalRouter());
    }
}
